cmake_minimum_required(VERSION 3.24)
project(npuzzle_player VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NP_BUILD_PLAYER "Build the SFML graphical player" ON)
option(NP_FETCH_SFML "Download and build the pinned SFML release" OFF)
include(CTest)
if(MSVC)
  add_compile_options(/utf-8 /W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_executable(path_check path_check.cpp)
add_executable(parser_tests tests/parser_tests.cpp)
add_executable(playback_tests tests/playback_tests.cpp)
target_include_directories(parser_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(playback_tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(BUILD_TESTING)
  add_test(NAME parser_contract COMMAND parser_tests)
  add_test(NAME playback_contract COMMAND playback_tests)
  add_test(NAME repository_examples COMMAND path_check "${CMAKE_CURRENT_SOURCE_DIR}/../examples/paths")
  add_test(NAME zero_move_fixture COMMAND path_check "${CMAKE_CURRENT_SOURCE_DIR}/test_cases/test_01.txt")
  add_test(NAME unsolvable_fixture COMMAND path_check "${CMAKE_CURRENT_SOURCE_DIR}/test_cases/test_unsolvable.txt")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/empty-path.txt" "")
  add_test(NAME reject_empty_file COMMAND path_check "${CMAKE_CURRENT_BINARY_DIR}/empty-path.txt")
  set_tests_properties(reject_empty_file PROPERTIES WILL_FAIL TRUE)
endif()

if(NP_BUILD_PLAYER)
  if(NP_FETCH_SFML)
    include(FetchContent)
    # Some Windows process environments omit PROCESSOR_ARCHITECTURE. SFML 2.6.2
    # assumes CMAKE_SYSTEM_PROCESSOR is nonempty; derive it from MinGW itself.
    if(WIN32 AND NOT CMAKE_SYSTEM_PROCESSOR AND MINGW)
      execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -dumpmachine
        OUTPUT_VARIABLE NP_COMPILER_TARGET OUTPUT_STRIP_TRAILING_WHITESPACE)
      if(NP_COMPILER_TARGET MATCHES "^x86_64")
        set(CMAKE_SYSTEM_PROCESSOR AMD64)
      elseif(NP_COMPILER_TARGET MATCHES "^aarch64")
        set(CMAKE_SYSTEM_PROCESSOR ARM64)
      elseif(NP_COMPILER_TARGET MATCHES "^i[3-6]86")
        set(CMAKE_SYSTEM_PROCESSOR x86)
      endif()
    endif()
    set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_DOC OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    # Immutable upstream commit for SFML 2.6.2, resolved from its release tag.
    FetchContent_Declare(sfml
      URL https://codeload.github.com/SFML/SFML/tar.gz/5383d2b3948f805af55c9f8a4587ac72ec5981d1
      URL_HASH SHA256=d10569c2b62809ac08fee6e5d7c403aca439eb36586332258c1a1279d0c1a669
      TIMEOUT 120
      DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(sfml)
  else()
    find_package(SFML 2.6.2 EXACT COMPONENTS graphics window system REQUIRED)
  endif()
  add_executable(npuzzle_player main.cpp)
  target_link_libraries(npuzzle_player PRIVATE sfml-graphics sfml-window sfml-system)
endif()

# Copy the existing MinGW runtimes beside executables; do not alter system PATH.
if(MINGW)
  get_filename_component(NP_COMPILER_BIN "${CMAKE_CXX_COMPILER}" DIRECTORY)
  foreach(NP_TARGET path_check parser_tests playback_tests npuzzle_player)
    if(TARGET ${NP_TARGET})
      foreach(NP_RUNTIME libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll)
        if(EXISTS "${NP_COMPILER_BIN}/${NP_RUNTIME}")
          add_custom_command(TARGET ${NP_TARGET} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
              "${NP_COMPILER_BIN}/${NP_RUNTIME}" "$<TARGET_FILE_DIR:${NP_TARGET}>")
        endif()
      endforeach()
    endif()
  endforeach()
endif()
